\section{Introduction}
\begin{frame}{Controller Area Network (CAN) -- Key Attributes and Benefits}
	\begin{columns}[c]
		\canbusanchor
		\begin{column}{0.40\textwidth}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item \textbf{Simple and low cost:} Single shared multi-master bus, widely used in automotive and industrial contexts
				\item \textbf{Robust:} Differential signaling, built-in error detection, faulty nodes isolated
				\item \textbf{Priority-based access:} Highest priority ID wins bit-wise arbitration
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{5pt}
		\item \textbf{Context:} Bosch, 1980s. ISO 11898-1. Each node = ECU (CAN controller + transceiver). Everllence: marine engine control on IO-extender FPGA.
		\item \textbf{Robustness:} CANH/CANL differential --- noise couples equally to both wires and cancels. Bus-off isolates faulty nodes automatically.
		\item \textbf{Arbitration:} Non-destructive --- losing node backs off without corrupting the winning frame. No collision recovery overhead.
		\item \textbf{Q: Why CAN over Ethernet or similar?} Hard real-time priority arbitration, deterministic latency, and far simpler embedded hardware.
		\item \textbf{Q: Why design in-house?} Covered on the Objectives slide.
	\end{itemize}}

% SLIDE: Bit Timing
% ==========================================
\begin{frame}{CAN Bit Timing and Synchronization}
	\begin{columns}[c]
		\begin{column}{0.42\textwidth}
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item Each bit: SYNC\_SEG, PROP\_SEG, PHASE\_SEG1, PHASE\_SEG2 --- sample point at end of PHASE\_SEG1
				\item Hard synchronization on SOF dominant edge; resynchronization on any recessive-to-dominant edge mid-frame
				\item CAN FD: arbitration phase at nominal rate, data phase at up to 8\,Mbit/s --- switch at BRS bit
				\item TDC measures transceiver loop delay on reserved bit; positions SSP for bit-error monitoring in data phase
			\end{itemize}
		\end{column}
		\begin{column}{0.58\textwidth}
			\begin{center}
				\includegraphics[width=\textwidth,height=0.38\textheight,keepaspectratio]{\figures/bit_timing.png}\\[0.4em]
				\includegraphics[width=\textwidth,height=0.38\textheight,keepaspectratio]{\figures/tdc.png}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{5pt}
		\item \textbf{Segments:} SYNC\_SEG = 1\,TQ (edge expected here). PROP\_SEG compensates bus propagation. PHASE\_SEG1/2 are adjustable for sync. Total bit time is configurable in time quanta (TQ).
		\item \textbf{Hard sync:} only on SOF, resets the bit timing counter. Resync: adjusts PHASE\_SEG1 (lengthen) or PHASE\_SEG2 (shorten) by up to the sync jump width.
		\item \textbf{Dual rate:} arbitration must be slow enough for all nodes to participate simultaneously (round-trip propagation). Data phase is point-to-point after arbitration --- can be much faster.
		\item \textbf{TDC:} transmitter drives a bit, echos it back via the transceiver. At 5\,Mbit/s a 200\,ns loop delay = 1 full bit period --- SP would sample the wrong bit. TDC counts from BRS edge to reserved bit echo, stores the offset, fires SSP at that offset each data-phase bit.
		\item \textbf{Q: Why not just use SP for bit-error monitoring in data phase?} SP fires at the nominal sample point --- at fast rates, the TX echo arrives late, so the transmitter would read its own bit one period behind. SSP fires at the correct offset.
	\end{itemize}}

% ==========================================
% SLIDE: Error Detection and Confinement
% ==========================================
\begin{frame}{Error Detection and Confinement}
	\begin{columns}[c]
		\begin{column}{0.42\textwidth}
			\textbf{Five error types}
			\begin{itemize}\setlength{\itemsep}{4pt}
				\item \textbf{Bit:} transmitted bit $\neq$ monitored bus level
				\item \textbf{Stuff:} six consecutive identical bits (transmitter inserts an opposite bit; receiver removes it)
				\item \textbf{CRC:} computed vs.\ received remainder mismatch
				\item \textbf{Form:} fixed-form field violated
				\item \textbf{ACK:} no dominant ACK received
			\end{itemize}
			\vspace{0.5em}
			Any detecting node transmits an error flag, destroying the current frame
		\end{column}
		\begin{column}{0.58\textwidth}
			\begin{center}
				\includegraphics[width=\textwidth,height=0.72\textheight,keepaspectratio]{\figures/fce_fsm.png}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{5pt}
		\item \textbf{Active error flag:} 6 dominant bits --- violates stuff rule, all nodes detect it and join in. Passive error flag: 6 recessive bits --- invisible on bus, other nodes continue.
		\item \textbf{TEC/REC rules:} TEC $+8$ on TX error, $-1$ on success. REC $+1$ on RX error, $-8$ on primary error (node caused it), $-1$ on success.
		\item \textbf{Error passive:} TEC or REC $\geq 128$. Node still participates but cannot destroy frames --- passive flags are recessive and ignored by other nodes.
		\item \textbf{Bus off:} TEC $\geq 256$. Node stops driving. Recovery: 128 sequences of 11 consecutive recessive bits.
		\item \textbf{Q: Why does error passive node still transmit?} It can still receive and ACK --- only its ability to signal errors is degraded. Useful for a node recovering from transient faults.
	\end{itemize}}

% SLIDE: CAN vs CAN FD
% ==========================================
\begin{frame}{CAN Classic (CC) vs. CAN Flexible Data Rate (CF)}
	\vspace{0.3em}
	\includegraphics[width=\textwidth,keepaspectratio]{images/frame_format.png}\\[0.3em]
	\footnotesize
	\renewcommand{\arraystretch}{1.5}
	\begin{tabular}{>{\bfseries}p{0.26\textwidth} p{0.26\textwidth} p{0.26\textwidth}}
		\toprule
		Feature                      & CC                               & CF                           \\
		\midrule
		Max bit rate                 & 1\,Mbit/s                        & 8\,Mbit/s (data phase)       \\
		Max payload                  & 8\,bytes                         & 64\,bytes                    \\
		Error detection HD$^\dagger$ & 6 typical, \textbf{2 worst case} & 6, all patterns, $\leq$64\,B \\
		Burst error detection        & up to 15\,bits                   & up to 17/21\,bits            \\
		\bottomrule
	\end{tabular}\\[0.2em]
	{\tiny $^\dagger$ CAN in Automation (CiA), \textit{CRC in CAN Frames}, can-cia.org.}
\end{frame}
\note{
\footnotesize\setlength{\parskip}{3pt}
\textbf{Dual bit rate:} Arbitration at nominal rate (backward compatible). Data phase
switches to fast rate at the BRS bit. TDC compensates transceiver loop delay.

\textbf{HD=6 requires fixed length:} CRC-15 BCH guarantees HD=6 only for fixed codeword
length. CAN Classic stuff bits make wire length variable --- receiver determines
stuff positions by counting, not from metadata.

\textbf{HD=2 worst case:} Two errors can \emph{relocate} a stuff bit (one deletes it,
one creates it elsewhere). Receiver strips the wrong bit, feeds a different logical
sequence to CRC. For specific data patterns the CRC accidentally matches --- undetected.
CAN FD fixes this: dynamic stuff bits included in CRC feed + SBC field (Gray-coded
count + parity) means any relocation changes the CRC remainder.

\textbf{Burst detection:} Burst error of length $b$ = polynomial $x^i \cdot B(x)$,
$\deg(B)=b{-}1$. Undetected only if $G(x)$ divides $B(x)$. Impossible when
$\deg(B) < \deg(G) = 15$. CRC-17/21 raise ceiling to 17/21 bits for longer FD frames.
}

\section{Design and Implementation}
\begin{frame}{Requirements Extraction -- ISO\,11898-1 to Structured Requirements}
	\begin{columns}[c]
		\begin{column}{0.50\textwidth}
			\begin{itemize}\setlength{\itemsep}{8pt}
				\item 38 requirements distilled from ISO\,11898-1 --- AI-assisted extraction, manual consolidation
				\item The ISO sub-layer structure became the module structure directly --- \texttt{layer} is the single field that maps each requirement to one implementation unit
				\item Classifying each requirement by \texttt{layer} and \texttt{priority} forced explicit design decisions before any RTL was written --- which module owns which behavior, and what is strictly needed vs.\ optional
				\item \textbf{Implementation order:} started at the protocol core (\texttt{can\_mac\_fsm}) and worked outward --- BS/CRC/SER, then PCS, then FCE, then integration
			\end{itemize}
		\end{column}
		\begin{column}{0.50\textwidth}
			\footnotesize
			\renewcommand{\arraystretch}{1.4}
			\begin{tabular}{>{\ttfamily}l p{0.60\textwidth}}
				\toprule
				id             & REQ-NNN                        \\
				source\_clause & ISO\,11898-1 section reference \\
				paraphrase     & Concise restatement            \\
				layer          & LLC / MAC / PCS / FCE / System \\
				priority       & P1 / P2 / P3                   \\
				\bottomrule
			\end{tabular}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{5pt}
		\item \textbf{Extraction:} Claude bootstrapped 168 normative statements from ISO\,11898-1. Manual distillation to 38 prioritised, independently verifiable requirements was substantial --- AI helped consistency, not speed.
		\item \textbf{Layer as design driver:} one ISO sub-layer = one RTL module. \texttt{layer} is the single field that produced the module split --- all other classification fields shape testbenches, not modules.
		\item \textbf{Implementation order:} MAC FSM is the protocol core --- everything interfaces to it. Sub-modules (BS, CRC, SER) built next as they have clear, narrow contracts with FSM. PCS below (provides SP/SSP strobes), FCE beside (monitors events). Integration last.
		\item \textbf{Q: Why MAC FSM first?} It defines the interface contract for every other module --- building anything else first would mean designing against a moving target.
		\item \textbf{Q: What does priority mean?} P1 = strictly needed for a correct implementation (shall clauses). P2 = nice-to-have. P3 = optional. A scope clarity tool, not a schedule.
	\end{itemize}}

\begin{frame}{From ISO Reference Model to RTL Modules}
	\begin{columns}[c]
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[height=0.75\textheight,keepaspectratio]{images/ecu.png}\\
				{\tiny ISO\,11898-1 reference model}
			\end{center}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[height=0.75\textheight,keepaspectratio]{images/can_controller.png}\\
				{\tiny Implemented RTL modules}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item ISO 11898-1 defines four sub-layers: LLC, MAC, PCS, and FCE. Each has a specific responsibility boundary --- the standard does not prescribe RTL structure, just the functional split.
		\item \textbf{LLC} (Logic Link Control): acceptance filtering on received frames, retransmission on arbitration loss or error, conversion between host-facing and MAC-facing frame formats.
		\item \textbf{MAC} (Medium Access Control): orchestrates frame TX and RX across all four formats, controls bit stuffing/destuffing, drives CRC computation, and commits bits to the bus.
		\item \textbf{PCS} (Physical Coding Sub-layer): bit timing, sample point generation, hard synchronization and resynchronization, TDC for SSP positioning in the data phase.
		\item \textbf{FCE} (Fault Confinement Entity): tracks TX and RX error counts, manages error active/error passive/bus off state transitions. Sits outside the data path --- monitors every TX and RX event as a cross-cutting entity.
		\item Right diagram: one RTL module per sub-layer. LLC is not implemented (out of scope). MAC is split into four sub-modules: \texttt{can\_mac\_fsm} orchestrates, \texttt{can\_mac\_ser} serializes the TX frame, \texttt{can\_mac\_bs} handles bit stuffing/destuffing, \texttt{can\_mac\_crc} runs three CRC engines in parallel.
		\item Key payoff: module boundaries = testbench boundaries. Each module has exactly one testbench. Requirements trace unambiguously to one module.
	\end{itemize}}

% SLIDE MAC-1: FSM per-field structure
% ==========================================
\begin{frame}{MAC: \texttt{can\_mac\_fsm} -- FSM orchestrating the MAC layer}
	\begin{columns}[c]
		\begin{column}{0.60\textwidth}
			\begin{itemize}\setlength{\itemsep}{8pt}
				\item CB, CE, FB and FE gets one branch each in the FSM state graph -- one state per frame field
				\item The \texttt{is\_transmitter} flag gates TX vs.\ RX logic within each state
				\item State transition on the sample point strobe from \texttt{can\_pcs}. Next bit is driven two cycles later once \texttt{can\_mac\_bs} and \texttt{can\_mac\_crc} outputs have settled
			\end{itemize}
		\end{column}
		\begin{column}{0.40\textwidth}
			\begin{center}
				\includegraphics[,height=0.85\textheight,keepaspectratio]{images/fsm.png}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item Point to the frame format image: CC and CF are identical up to FDF. From FDF onward the field sequences diverge --- BRS, ESI, longer data, SBC, different CRC. Each of those fields is one FSM state.
		\item Format transitions (e.g. CC to CF at FDF bit) are state graph edges, not conditionals inside a shared counting state. This makes the FSM directly readable against the ISO field sequence table.
		\item \texttt{is\_transmitter} is set at frame start and held for the duration. In each state, TX logic (driving the bus, reading from \texttt{can\_mac\_ser}) is gated on \texttt{is\_transmitter='1'}; RX logic (sampling the bus, writing to the RX buffer) is gated on \texttt{is\_transmitter='0'}. This is the mechanism that lets one FSM serve both directions without duplication.
		\item Traceability is structural --- each requirement maps to exactly one FSM state. Falls out of the one-state-per-field decision automatically, not a documentation exercise.
		\item Split-path attempt: TX and RX into separate FSMs --- \texttt{side} dimension misread as implementation-facing. Two weeks of parallel development produced eight timing divergences.
		\item Three representative bugs: SOF structural divergence (systematic, every FD frame); \texttt{in\_data\_phase} cleared 1~SP early at CRC boundary ($\sim$265 FD frames); ACK window excluded \texttt{s\_ack\_delimiter} (unreachable with zero-delay loopback).
		\item Unified path: a second independent CAN implementation verifying the first. Bug~D structurally impossible to trigger in a single-node testbench --- only visible with two nodes and realistic bus delay.
	\end{itemize}}

% SLIDE MAC-2: Bit stuffing
% ==========================================
\begin{frame}{MAC: \texttt{can\_mac\_bs} -- Bit Stuffing and Destuffing}
	\begin{columns}[c]
		\begin{column}{0.50\textwidth}
			\textbf{Dynamic stuffing} (CC and CF):
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item Transmitter inserts inverse bit after five consecutive identical bits
				\item Receiver strips stuff bits and checks for violations
			\end{itemize}
			\vspace{0.4em}
			\textbf{Fixed stuffing} (CF CRC region only):
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item Stuff bit at fixed intervals
				\item SBC: Gray-coded dynamic stuff count + parity
			\end{itemize}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[width=1.0\textwidth,keepaspectratio]{images/bs.png}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item \texttt{can\_mac\_bs} is a single shared instance -- same module handles TX stuffing and RX destuffing, switched by \texttt{is\_transmitter}.
		\item Dynamic stuffing applies from SOF through the data field in both CC and CF. Fixed stuffing applies only to the CF CRC region.
		\item SBC (Stuff Bit Count): Gray-coded count of dynamic stuff bits inserted since SOF, plus a parity bit. Transmitted before the CF CRC. Receiver verifies independently of CRC.
		\item HD=2 fix: in CC, dynamic stuff bits excluded from CRC feed --- two errors can relocate a stuff bit and leave CRC unchanged. In CF, dynamic stuff bits included in \texttt{data\_fd}, so any relocation changes the remainder.
	\end{itemize}}

% SLIDE MAC-3: CRC
% ==========================================
\begin{frame}{MAC: \texttt{can\_mac\_crc} -- CRC Engine}
	\begin{columns}[c]
		\begin{column}{0.50\textwidth}
			\textbf{Three parallel engines}:
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item CRC-15: CC frames
				\item CRC-17: CF frames, data payload $\leq$ 16\,bytes
				\item CRC-21: CF frames, data payload $>$ 16\,bytes
			\end{itemize}
			\vspace{0.4em}
			\textbf{Dual data feed}:
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item \texttt{data\_cc}: de-stuffed stream
				\item \texttt{data\_fd}: includes dynamic stuff bits
			\end{itemize}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[width=1.0\textwidth,keepaspectratio]{images/crc.png}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item All three CRC engines run every cycle. Output selected at CRC transmission/check time based on format and DLC.
		\item \texttt{data\_cc} is the de-stuffed stream, identical to what the CC receiver reconstructs. \texttt{data\_fd} includes dynamic stuff bits --- this restores the HD=6 guarantee lost in CC due to variable wire length.
		\item CRC-17 and CRC-21 are both BCH codes with HD=6 guarantee for fixed-length codewords. CF fixed stuffing makes wire-length predictable, restoring the HD=6 property lost in CC.
	\end{itemize}}

\begin{frame}{MAC: \texttt{can\_mac\_ser} -- TX Frame Serializer}
	\begin{columns}[c]
		\begin{column}{0.70\textwidth}
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item Converts LLC byte stream to serial bit stream for \texttt{can\_mac\_fsm}
				\item Receives frame from LLC over Avalon-ST interface
				\item Extracts frame metadata (IDE, FDF, DLC, FTYP, BRS, ESI) from two leading config bytes in LLC frame
			\end{itemize}
		\end{column}
		\begin{column}{0.30\textwidth}
			\begin{center}
				\includegraphics[height=0.90\textheight,keepaspectratio]{images/ser.png}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item \texttt{can\_mac\_ser} is TX-only --- there is no RX path. The receiver samples bits directly from the bus via \texttt{can\_pcs}.
		\item The LLC-MAC frame format front-loads all metadata into two leading config bytes, so \texttt{can\_mac\_ser} can begin streaming ID and data bits immediately without buffering the full 71-byte frame.
		\item Padding bits in the 32-bit ID field: base ID occupies bits [31:21] (11 bits, 21 trailing unused); extended ID occupies bits [31:3] (29 bits, 3 trailing unused). \texttt{can\_mac\_ser} skips these without asserting valid.
		\item \texttt{transfer\_status} from \texttt{can\_mac\_fsm} is forwarded back to \texttt{can\_llc} --- on any non-ongoing status (transmitted, aborted, lost arbitration) \texttt{can\_mac\_ser} returns to its reset state.
	\end{itemize}}

% SLIDE MAC-5: FCE
% ==========================================
\begin{frame}{FCE: \texttt{can\_fce} -- Fault Confinement Entity}
	\begin{columns}[c]
		\begin{column}{0.70\textwidth}
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item Three states: error active, error passive, bus off. Driven by TEC and REC thresholds
				\item Subscribes to error events from \texttt{can\_mac\_fsm}. Signals \texttt{error\_active} and \texttt{bus\_off} back
				\item Bus-off recovery after 128 \texttt{idle\_condition} strobes from \texttt{can\_pcs}
			\end{itemize}
		\end{column}
		\begin{column}{0.30\textwidth}
			\begin{center}
				\includegraphics[height=0.90\textheight,keepaspectratio]{images/fce.png}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{5pt}
		\item \textbf{Error counters:} TEC $+8$ on TX error, $-1$ on successful TX. REC $+1$ on RX error, $-1$ on successful RX.
		\item \textbf{State thresholds:} Error active: TEC and REC $<128$. Error passive: TEC or REC $\geq128$ --- passive flags are recessive, invisible on bus. Bus off: TEC $\geq256$ --- stops transmitting.
		\item \textbf{Error events:} \texttt{primary\_error}, \texttt{sending\_error\_flag}, \texttt{passive\_tx\_ack\_error}, \texttt{error\_delim\_to\_late}, \texttt{successful\_transfer}.
		      \begin{itemize}\setlength{\itemsep}{1pt}
			      \item \texttt{passive\_tx\_ack\_error}: ACK failure for an error-passive transmitter -- ISO exempts this from TEC increment to avoid stranding lone node at bus off.
			      \item \texttt{error\_delim\_to\_late}: delimiter window expired with no recessive edge --- stuck-dominant fault. Hits both TEC and REC by 8 to fast-track bus off.
		      \end{itemize}
		\item \textbf{Q: Why 128 sequences?} ISO mandates it -- extended quiet period prevents rapid re-entry after bus-off.
		\item \textbf{Q: In-flight frame on bus off?} \texttt{can\_mac\_fsm} receives \texttt{bus\_off} and aborts immediately.
	\end{itemize}}

% ==========================================
% SLIDE 6: PCS
% ==========================================
\begin{frame}{PCS: \texttt{can\_pcs} -- Bit Timing and Transmitter Delay Compensation (TDC)}
	\begin{columns}[c]
		\begin{column}{0.50\textwidth}
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item SP strobes at end of PHASE\_SEG1
				\item Transmit bit at end of PHASE\_SEG2
				\item SSP strobe used for bit-error monitoring in data phase
				\item Switch to data phase bit rate at BRS bit
				\item Transceiver loop delay measured on res bit
			\end{itemize}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[width=\textwidth,height=0.75\textheight,keepaspectratio]{images/pcs.png}\\[0.3em]
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item Nominal rate: e.g.\ 500\,kbit/s for arbitration. Data phase: up to 5\,Mbit/s, entered at BRS bit, exited at CRC delimiter.
		\item Hard synchronization: on SOF dominant edge after bus idle --- resets the bit timing counter. Resynchronization: on any recessive-to-dominant edge during a frame --- adjusts PHASE\_SEG1 or PHASE\_SEG2 by the sync jump width.
		\item TDC: the transceiver introduces a loop delay between TX drive and RX echo. At fast data rates this is a significant fraction of the bit period. TDC measures it on the reserved bit, counts down from BRS, and fires SSP at the correct offset.
		\item SSP fires independently of SP --- transmitter uses SSP for bit-error detection in data phase, not SP.
	\end{itemize}}

			height=0.70\textheight,

	\end{center}

	{\footnotesize
		Error active $\to$ error passive (TEC\,=\,128) $\to$
		bus off (TEC\,=\,256) $\to$ 128 idle sequences $\to$ error active}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{5pt}
		\item \textbf{Waveform:} Each TEC increment driven by injected errors from \texttt{p\_test\_ctrl}. TEC crosses 128 (error passive), crosses 256 (bus off), then 128 idle sequences counted by \texttt{can\_pcs} before recovery.
		\item \textbf{Recovery mechanism:} \texttt{can\_pcs} emits \texttt{idle\_condition} strobes on 11 consecutive recessive bits. FCE counts 128 of these, then transitions back to error active and signals \texttt{can\_mac\_fsm}.
		\item \textbf{Integration evidence:} This requires FCE, PCS, and MAC FSM all operating correctly together --- a unit testbench cannot produce this sequence.
		\item \textbf{Q: Why 128 sequences of 11 recessive bits?} ISO 11898-1 mandates it --- extended quiet period ensures the bus is genuinely idle before the node re-joins.
		\item \textbf{Q: Can a node re-enter bus off immediately after recovery?} Yes --- if errors continue, TEC climbs back to 256 and the cycle repeats.
	\end{itemize}}
