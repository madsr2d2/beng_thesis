\section{Introduction}
\begin{frame}{Controller Area Network (CAN) -- Bus Topology and ECUs}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\begin{center}
				\vspace{1.2em}
				\includegraphics[width=\textwidth,height=0.65\textheight,keepaspectratio]{images/bus_wire.jpeg}\\[0.3em]
				{\tiny CAN bus wire - notice the twist, (source: Amazon.com)}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{
	\footnotesize
	\begin{itemize}\setlength{\itemsep}{4pt}
		\item CAN = Controller Area Network. Bosch, 1980s. Now ISO 11898-1 --- the dominant embedded bus standard.
		\item Each box = an ECU (Electronic Control Unit). CAN controller + transceiver chip. No host, no bus master.
		\item Used in: cars, trucks, marine, industrial machinery, aerospace, agriculture.
		\item Everllence: marine engine control. Sensors and actuators on one shared CAN bus on the IO-extender FPGA.
	\end{itemize}
}

% ==========================================
% SLIDE 2: Benefit 1
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\vspace{1.2em}
			\textbf{Simple and Low Cost}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Single shared bus replaces complex point-to-point wiring
				\item Reduces vehicle/machine weight, wiring errors, and costs
				\item Widely adopted: low-cost controllers, transceivers, and tooling
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Good example: NMEA 0183 (point-to-point, one talker per line) vs NMEA 2000
	(CAN-based, all devices on one bus) in marine systems --- exactly Everllence's world.
	Weight reduction in automotive: switching to CAN can save up to 20 kg of wiring harness.
}

% ==========================================
% SLIDE 3: Benefit 2
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\vspace{1.2em}
			\textbf{Easy Centralized Access}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item One connection point for diagnostics across all ECUs on the network
				\item Silent data logging --- the interface does not affect bus traffic
				\item Remote ECU firmware and configuration updates via CAN frames
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Silent mode is critical for non-invasive diagnostics in production systems ---
	a logging device that influences the bus would invalidate the measurement.
	ECU flashing over CAN is standard in automotive (UDS protocol) and is used
	in Everllence's engine platforms too.
}

% ==========================================
% SLIDE 4: Benefit 3
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\vspace{1.2em}
			\textbf{Robust Against Interference}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Differential signaling (CANH/CANL): noise couples equally to both wires and cancels
				\item Built-in error detection: bit, stuff, CRC, form, and ACK errors
				\item Faulty nodes automatically isolated --- bus off prevents one node corrupting the network
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	The marine engine bay is one of the harshest EMI environments imaginable ---
	high-current motor drives, ignition systems, inverters. Differential signaling
	and bus-off isolation are not nice-to-haves here, they are requirements.
	Bus off is implemented in the FCE (Fault Confinement Entity) in this project ---
	good segue to the architecture slides later.
}

% ==========================================
% SLIDE 5: Benefit 4
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\vspace{1.2em}
			\textbf{Efficient Priority-Based Access}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Bitwise arbitration: lowest identifier wins, no collisions, no bus master
				\item Safety-critical messages always transmitted first
				\item Lower-priority frames automatically fill remaining bus bandwidth
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Non-destructive arbitration: the losing node does not corrupt the winning frame ---
	it simply backs off and retries. No wasted bandwidth, no collision recovery overhead.
	This is the key reason CAN outperformed competing protocols like VAN and SCP
	in the 1990s for safety-critical automotive use.
}

% ==========================================
% SLIDE 3: CAN vs CAN FD
% ==========================================
\begin{frame}{CAN Classic (CC) vs. CAN Flexible Data Rate (CF)}
	\vspace{0.3em}
	\includegraphics[width=\textwidth,height=0.38\textheight,keepaspectratio]{images/frame_format.png}\\[0.3em]
	\footnotesize
	\renewcommand{\arraystretch}{1.5}
	\begin{tabular}{>{\bfseries}p{0.40\textwidth} p{0.26\textwidth} p{0.26\textwidth}}
		\toprule
		Feature                      & CAN Classic                      & CAN FD                       \\
		\midrule
		Max bit rate                 & 1\,Mbit/s                        & 8\,Mbit/s (data phase)       \\
		Max payload                  & 8\,bytes                         & 64\,bytes                    \\
		Error detection HD$^\dagger$ & 6 typical, \textbf{2 worst case} & 6, all patterns, $\leq$64\,B \\
		Burst error detection        & up to 15\,bits                   & up to 17/21\,bits            \\
		\bottomrule
	\end{tabular}\\[0.2em]
	{\tiny $^\dagger$ HD degrades to 2 for specific bit patterns due to stuff-bit cancellation.
	Source: CAN in Automation (CiA), \textit{CRC in CAN Frames}, can-cia.org.}
\end{frame}
\note{
\footnotesize\setlength{\parskip}{3pt}
\textbf{Dual bit rate:} Arbitration at nominal rate (backward compatible). Data phase
switches to fast rate at the BRS bit. TDC compensates transceiver loop delay.

\textbf{HD=6 requires fixed length:} CRC-15 BCH guarantees HD=6 only for fixed codeword
length. CAN Classic stuff bits make wire length variable --- receiver determines
stuff positions by counting, not from metadata.

\textbf{HD=2 worst case:} Two errors can \emph{relocate} a stuff bit (one deletes it,
one creates it elsewhere). Receiver strips the wrong bit, feeds a different logical
sequence to CRC. For specific data patterns the CRC accidentally matches --- undetected.
CAN FD fixes this: dynamic stuff bits included in CRC feed + SBC field (Gray-coded
count + parity) means any relocation changes the CRC remainder.

\textbf{Burst detection:} Burst error of length $b$ = polynomial $x^i \cdot B(x)$,
$\deg(B)=b{-}1$. Undetected only if $G(x)$ divides $B(x)$. Impossible when
$\deg(B) < \deg(G) = 15$. CRC-17/21 raise ceiling to 17/21 bits for longer FD frames.
}

\section{Design and Implementation}
\begin{frame}{From ISO Reference Model to RTL Modules}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{center}
				\includegraphics[width=1.0\textwidth,height=0.75\textheight,keepaspectratio]{images/ecu.png}\\[0.3em]
				{\small ISO\,11898-1 reference model}
			\end{center}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{center}
				\includegraphics[width=1.0\textwidth,height=0.75\textheight,keepaspectratio]{images/can_controller.png}\\[0.3em]
				{\small Implemented RTL modules}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item ISO 11898-1 defines four sub-layers: LLC, MAC, PCS, and FCE. Each has a specific responsibility boundary --- the standard does not prescribe RTL structure, just the functional split.
		\item \textbf{LLC} (Logic Link Control): acceptance filtering on received frames, retransmission on arbitration loss or error, conversion between host-facing and MAC-facing frame formats.
		\item \textbf{MAC} (Medium Access Control): orchestrates frame TX and RX across all four formats, controls bit stuffing/destuffing, drives CRC computation, and commits bits to the bus.
		\item \textbf{PCS} (Physical Coding Sub-layer): bit timing, sample point generation, hard synchronization and resynchronization, TDC for SSP positioning in the data phase.
		\item \textbf{FCE} (Fault Confinement Entity): tracks TX and RX error counts, manages error active/error passive/bus off state transitions. Sits outside the data path --- monitors every TX and RX event as a cross-cutting entity.
		\item Right diagram: one RTL module per sub-layer. LLC is not implemented (out of scope). MAC is split into four sub-modules: \texttt{can\_mac\_fsm} orchestrates, \texttt{can\_mac\_ser} serializes the TX frame, \texttt{can\_mac\_bs} handles bit stuffing/destuffing, \texttt{can\_mac\_crc} runs three CRC engines in parallel.
		\item Key payoff: module boundaries = testbench boundaries. Each module has exactly one testbench. Requirements trace unambiguously to one module.
	\end{itemize}}

% ==========================================
% SLIDE Design-2: MAC FSM Granularity
% ==========================================
\begin{frame}{MAC FSM: One State per Frame Field}
	\begin{columns}[T]
		\begin{column}{0.50\textwidth}
			\textbf{Design decision}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item One \texttt{can\_mac\_fsm} state per MAC frame field
				\item Format divergences (CC vs.\ CF at the FDF bit) become state graph edges --- not counter conditionals inside a shared state
			\end{itemize}
		\end{column}
		\begin{column}{0.46\textwidth}
			\textbf{Consequence}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Each frame field requirement maps to exactly one FSM state
				\item Traceability is structural --- not a documentation exercise
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item Split-path attempt: TX and RX into separate FSMs --- \texttt{side} dimension misread as implementation-facing. Two weeks of parallel development produced eight timing divergences.
		\item Three representative bugs: SOF structural divergence (systematic, every FD frame); \texttt{in\_data\_phase} cleared 1~SP early at CRC boundary ($\sim$265 FD frames); ACK window excluded \texttt{s\_ack\_delimiter} (unreachable with zero-delay loopback).
		\item Unified path: a second independent CAN implementation verifying the first. Bug~D is structurally impossible to trigger in a single-node testbench --- only visible with two nodes and realistic bus delay.
	\end{itemize}}
