\section{Introduction}
\begin{frame}{Controller Area Network (CAN) -- Bus Topology and ECUs}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.40\textwidth}
			\begin{center}
				\vspace{1.2em}
				\includegraphics[width=\textwidth,height=0.65\textheight,keepaspectratio]{images/bus_wire.jpeg}\\[0.3em]
				{\tiny CAN bus wire - notice the twist, (source: Amazon.com)}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{
	\footnotesize
	\begin{itemize}\setlength{\itemsep}{4pt}
		\item CAN = Controller Area Network. Bosch, 1980s. Now ISO 11898-1 --- the dominant embedded bus standard.
		\item Each box = an ECU (Electronic Control Unit). CAN controller + transceiver chip. No host, no bus master.
		\item Used in: cars, trucks, marine, industrial machinery, aerospace, agriculture.
		\item Everllence: marine engine control. Sensors and actuators on one shared CAN bus on the IO-extender FPGA.
	\end{itemize}
}

% ==========================================
% SLIDE 2: Benefit 1
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.40\textwidth}
			\vspace{1.2em}
			\textbf{Simple and Low Cost}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Single shared bus replaces complex point-to-point wiring
				\item Reduces vehicle/machine weight, wiring errors, and costs
				\item Widely adopted: low-cost controllers, transceivers, and tooling
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Good example: NMEA 0183 (point-to-point, one talker per line) vs NMEA 2000
	(CAN-based, all devices on one bus) in marine systems --- exactly Everllence's world.
	Weight reduction in automotive: switching to CAN can save up to 20 kg of wiring harness.
}

% ==========================================
% SLIDE 3: Benefit 2
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.40\textwidth}
			\vspace{1.2em}
			\textbf{Easy Centralized Access}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item One connection point for diagnostics across all ECUs on the network
				\item Silent data logging --- the interface does not affect bus traffic
				\item Remote ECU firmware and configuration updates via CAN frames
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Silent mode is critical for non-invasive diagnostics in production systems ---
	a logging device that influences the bus would invalidate the measurement.
	ECU flashing over CAN is standard in automotive (UDS protocol) and is used
	in Everllence's engine platforms too.
}

% ==========================================
% SLIDE 4: Benefit 3
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.40\textwidth}
			\vspace{1.2em}
			\textbf{Robust Against Interference}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Differential signaling (CANH/CANL): noise couples equally to both wires and cancels
				\item Built-in error detection: bit, stuff, CRC, form, and ACK errors
				\item Faulty nodes automatically isolated --- bus off prevents one node corrupting the network
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	The marine engine bay is one of the harshest EMI environments imaginable ---
	high-current motor drives, ignition systems, inverters. Differential signaling
	and bus-off isolation are not nice-to-haves here, they are requirements.
	Bus off is implemented in the FCE (Fault Confinement Entity) in this project ---
	good segue to the architecture slides later.
}

% ==========================================
% SLIDE 5: Benefit 4
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.40\textwidth}
			\vspace{1.2em}
			\textbf{Efficient Priority-Based Access}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Bitwise arbitration: lowest identifier wins, no collisions, no bus master
				\item Safety-critical messages always transmitted first
				\item Lower-priority frames automatically fill remaining bus bandwidth
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Non-destructive arbitration: the losing node does not corrupt the winning frame ---
	it simply backs off and retries. No wasted bandwidth, no collision recovery overhead.
	This is the key reason CAN outperformed competing protocols like VAN and SCP
	in the 1990s for safety-critical automotive use.
}

% ==========================================
% SLIDE 3: CAN vs CAN FD
% ==========================================
\begin{frame}{CAN Classic (CC) vs. CAN Flexible Data Rate (CF)}
	\vspace{0.3em}
	\includegraphics[width=\textwidth,keepaspectratio]{images/frame_format.png}\\[0.3em]
	\footnotesize
	\renewcommand{\arraystretch}{1.5}
	\begin{tabular}{>{\bfseries}p{0.26\textwidth} p{0.26\textwidth} p{0.26\textwidth}}
		\toprule
		Feature                      & CC                               & CF                           \\
		\midrule
		Max bit rate                 & 1\,Mbit/s                        & 8\,Mbit/s (data phase)       \\
		Max payload                  & 8\,bytes                         & 64\,bytes                    \\
		Error detection HD$^\dagger$ & 6 typical, \textbf{2 worst case} & 6, all patterns, $\leq$64\,B \\
		Burst error detection        & up to 15\,bits                   & up to 17/21\,bits            \\
		\bottomrule
	\end{tabular}\\[0.2em]
	{\tiny $^\dagger$ CAN in Automation (CiA), \textit{CRC in CAN Frames}, can-cia.org.}
\end{frame}
\note{
\footnotesize\setlength{\parskip}{3pt}
\textbf{Dual bit rate:} Arbitration at nominal rate (backward compatible). Data phase
switches to fast rate at the BRS bit. TDC compensates transceiver loop delay.

\textbf{HD=6 requires fixed length:} CRC-15 BCH guarantees HD=6 only for fixed codeword
length. CAN Classic stuff bits make wire length variable --- receiver determines
stuff positions by counting, not from metadata.

\textbf{HD=2 worst case:} Two errors can \emph{relocate} a stuff bit (one deletes it,
one creates it elsewhere). Receiver strips the wrong bit, feeds a different logical
sequence to CRC. For specific data patterns the CRC accidentally matches --- undetected.
CAN FD fixes this: dynamic stuff bits included in CRC feed + SBC field (Gray-coded
count + parity) means any relocation changes the CRC remainder.

\textbf{Burst detection:} Burst error of length $b$ = polynomial $x^i \cdot B(x)$,
$\deg(B)=b{-}1$. Undetected only if $G(x)$ divides $B(x)$. Impossible when
$\deg(B) < \deg(G) = 15$. CRC-17/21 raise ceiling to 17/21 bits for longer FD frames.
}

\section{Design and Implementation}
\begin{frame}{From ISO Reference Model to RTL Modules}
	\begin{columns}[T]
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[height=0.75\textheight,keepaspectratio]{images/ecu.png}\\
				{\tiny ISO\,11898-1 reference model}
			\end{center}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[height=0.75\textheight,keepaspectratio]{images/can_controller.png}\\
				{\tiny Implemented RTL modules}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item ISO 11898-1 defines four sub-layers: LLC, MAC, PCS, and FCE. Each has a specific responsibility boundary --- the standard does not prescribe RTL structure, just the functional split.
		\item \textbf{LLC} (Logic Link Control): acceptance filtering on received frames, retransmission on arbitration loss or error, conversion between host-facing and MAC-facing frame formats.
		\item \textbf{MAC} (Medium Access Control): orchestrates frame TX and RX across all four formats, controls bit stuffing/destuffing, drives CRC computation, and commits bits to the bus.
		\item \textbf{PCS} (Physical Coding Sub-layer): bit timing, sample point generation, hard synchronization and resynchronization, TDC for SSP positioning in the data phase.
		\item \textbf{FCE} (Fault Confinement Entity): tracks TX and RX error counts, manages error active/error passive/bus off state transitions. Sits outside the data path --- monitors every TX and RX event as a cross-cutting entity.
		\item Right diagram: one RTL module per sub-layer. LLC is not implemented (out of scope). MAC is split into four sub-modules: \texttt{can\_mac\_fsm} orchestrates, \texttt{can\_mac\_ser} serializes the TX frame, \texttt{can\_mac\_bs} handles bit stuffing/destuffing, \texttt{can\_mac\_crc} runs three CRC engines in parallel.
		\item Key payoff: module boundaries = testbench boundaries. Each module has exactly one testbench. Requirements trace unambiguously to one module.
	\end{itemize}}

% ==========================================
% SLIDE MAC-1: FSM per-field structure
% ==========================================
\begin{frame}{MAC: \texttt{can\_mac\_fsm} -- FSM orchestrating the MAC layer}
	\begin{columns}[T]
		\begin{column}{0.50\textwidth}
			\begin{itemize}\setlength{\itemsep}{8pt}
				\item CB, CE, FB and FE gets one branch each in the FSM state graph -- one state per frame field
				\item The \texttt{is\_transmitter} flag gates TX vs.\ RX logic within each state
				\item State transition on the sample point strobe from \texttt{can\_pcs}. Next bit is driven two cycles later once \texttt{can\_mac\_bs} and \texttt{can\_mac\_crc} outputs have settled
			\end{itemize}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[width=\textwidth,height=0.70\textheight,keepaspectratio]{images/fsm.png}\\[0.3em]
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item Point to the frame format image: CC and CF are identical up to FDF. From FDF onward the field sequences diverge --- BRS, ESI, longer data, SBC, different CRC. Each of those fields is one FSM state.
		\item Format transitions (e.g. CC to CF at FDF bit) are state graph edges, not conditionals inside a shared counting state. This makes the FSM directly readable against the ISO field sequence table.
		\item \texttt{is\_transmitter} is set at frame start and held for the duration. In each state, TX logic (driving the bus, reading from \texttt{can\_mac\_ser}) is gated on \texttt{is\_transmitter='1'}; RX logic (sampling the bus, writing to the RX buffer) is gated on \texttt{is\_transmitter='0'}. This is the mechanism that lets one FSM serve both directions without duplication.
		\item Traceability is structural --- each requirement maps to exactly one FSM state. Falls out of the one-state-per-field decision automatically, not a documentation exercise.
		\item Split-path attempt: TX and RX into separate FSMs --- \texttt{side} dimension misread as implementation-facing. Two weeks of parallel development produced eight timing divergences.
		\item Three representative bugs: SOF structural divergence (systematic, every FD frame); \texttt{in\_data\_phase} cleared 1~SP early at CRC boundary ($\sim$265 FD frames); ACK window excluded \texttt{s\_ack\_delimiter} (unreachable with zero-delay loopback).
		\item Unified path: a second independent CAN implementation verifying the first. Bug~D structurally impossible to trigger in a single-node testbench --- only visible with two nodes and realistic bus delay.
	\end{itemize}}

% ==========================================
% SLIDE MAC-2: Bit stuffing
% ==========================================
\begin{frame}{MAC: \texttt{can\_mac\_bs} -- Bit Stuffing and Destuffing}
	\begin{columns}[T]
		\begin{column}{0.50\textwidth}
			\textbf{Dynamic stuffing} (CC and CF):
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item Transmitter inserts inverse bit after five consecutive identical bits
				\item Receiver strips stuff bits and checks for violations
			\end{itemize}
			\vspace{0.4em}
			\textbf{Fixed stuffing} (CF CRC region only):
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item Stuff bit at fixed intervals
				\item SBC: Gray-coded dynamic stuff count + parity
			\end{itemize}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[width=\textwidth,height=0.75\textheight,keepaspectratio]{images/bs.png}\\[0.3em]
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item \texttt{can\_mac\_bs} is a single shared instance -- same module handles TX stuffing and RX destuffing, switched by \texttt{is\_transmitter}.
		\item Dynamic stuffing applies from SOF through the data field in both CC and CF. Fixed stuffing applies only to the CF CRC region.
		\item SBC (Stuff Bit Count): Gray-coded count of dynamic stuff bits inserted since SOF, plus a parity bit. Transmitted before the CF CRC. Receiver verifies independently of CRC.
		\item HD=2 fix: in CC, dynamic stuff bits excluded from CRC feed --- two errors can relocate a stuff bit and leave CRC unchanged. In CF, dynamic stuff bits included in \texttt{data\_fd}, so any relocation changes the remainder.
	\end{itemize}}

% ==========================================
% SLIDE MAC-3: CRC
% ==========================================
\begin{frame}{MAC: \texttt{can\_mac\_crc} -- CRC Engine}
	\begin{columns}[T]
		\begin{column}{0.50\textwidth}
			\textbf{Three parallel engines}:
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item CRC-15: CC frames
				\item CRC-17: CF frames, data payload $\leq$ 16\,bytes
				\item CRC-21: CF frames, data payload $>$ 16\,bytes
			\end{itemize}
			\vspace{0.4em}
			\textbf{Dual data feed}:
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item \texttt{data\_cc}: de-stuffed stream
				\item \texttt{data\_fd}: includes dynamic stuff bits
			\end{itemize}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{center}
				\includegraphics[width=\textwidth,height=0.75\textheight,keepaspectratio]{images/crc.png}\\[0.3em]
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item All three CRC engines run every cycle. Output selected at CRC transmission/check time based on format and DLC.
		\item \texttt{data\_cc} is the de-stuffed stream, identical to what the CC receiver reconstructs. \texttt{data\_fd} includes dynamic stuff bits --- this restores the HD=6 guarantee lost in CC due to variable wire length.
		\item CRC-17 and CRC-21 are both BCH codes with HD=6 guarantee for fixed-length codewords. CF fixed stuffing makes wire-length predictable, restoring the HD=6 property lost in CC.
	\end{itemize}}

% ==========================================
% SLIDE MAC-4: Serializer
% ==========================================
\begin{frame}{MAC: \texttt{can\_mac\_ser} -- TX Frame Serializer}
	\begin{columns}[T]
		\begin{column}{0.40\textwidth}
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item Converts LLC byte stream to serial bit stream for \texttt{can\_mac\_fsm}
				\item Receives frame from LLC over Avalon-ST interface
				\item Extracts frame metadata (IDE, FDF, DLC, FTYP, BRS, ESI) from two leading config bytes in LLC frame
			\end{itemize}
		\end{column}
		\begin{column}{0.60\textwidth}
			\begin{center}
				\includegraphics[width=\textwidth,height=0.75\textheight,keepaspectratio]{images/ser.png}\\[0.3em]
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item \texttt{can\_mac\_ser} is TX-only --- there is no RX path. The receiver samples bits directly from the bus via \texttt{can\_pcs}.
		\item The LLC-MAC frame format front-loads all metadata into two leading config bytes, so \texttt{can\_mac\_ser} can begin streaming ID and data bits immediately without buffering the full 71-byte frame.
		\item Padding bits in the 32-bit ID field: base ID occupies bits [31:21] (11 bits, 21 trailing unused); extended ID occupies bits [31:3] (29 bits, 3 trailing unused). \texttt{can\_mac\_ser} skips these without asserting valid.
		\item \texttt{transfer\_status} from \texttt{can\_mac\_fsm} is forwarded back to \texttt{can\_llc} --- on any non-ongoing status (transmitted, aborted, lost arbitration) \texttt{can\_mac\_ser} returns to its reset state.
	\end{itemize}}

% ==========================================
% SLIDE 6: PCS
% ==========================================
\begin{frame}{PCS: \texttt{can\_pcs} -- Bit Timing and Transmitter Delay Compensation (TDC)}
	\begin{columns}[T]
		\begin{column}{0.40\textwidth}
			\begin{itemize}\setlength{\itemsep}{6pt}
				\item SP strobes at end of PHASE\_SEG1
				\item Transmit bit at end of PHASE\_SEG2
				\item SSP strobe used for bit-error monitoring in data phase
				\item Switch to data phase bit rate at BRS bit
				\item Transceiver loop delay measured on res bit
			\end{itemize}
		\end{column}
		\begin{column}{0.60\textwidth}
			\begin{center}
				\includegraphics[width=\textwidth,height=0.75\textheight,keepaspectratio]{images/pcs.png}\\[0.3em]
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item Nominal rate: e.g.\ 500\,kbit/s for arbitration. Data phase: up to 5\,Mbit/s, entered at BRS bit, exited at CRC delimiter.
		\item Hard synchronization: on SOF dominant edge after bus idle --- resets the bit timing counter. Resynchronization: on any recessive-to-dominant edge during a frame --- adjusts PHASE\_SEG1 or PHASE\_SEG2 by the sync jump width.
		\item TDC: the transceiver introduces a loop delay between TX drive and RX echo. At fast data rates this is a significant fraction of the bit period. TDC measures it on the reserved bit, counts down from BRS, and fires SSP at the correct offset.
		\item SSP fires independently of SP --- transmitter uses SSP for bit-error detection in data phase, not SP.
	\end{itemize}}

				side           & TX / RX / both                         \\
				format         & CB / CE / FB / FE                      \\
				observability  & black\_box / white\_box                \\
				method         & simulation / waveform / inspection     \\
				priority       & P1 / P2 / P3                           \\
				status         & not\_started / in\_progress / complete \\

		\end{column}

\note{\footnotesize\begin{itemize}\setlength{\itemsep}{4pt}
		\item AI-assisted extraction: Claude bootstrapped 168 normative statements from ISO\,11898-1. Manual distillation step --- consolidating to 37 prioritised, independently verifiable requirements --- was substantial and irreplaceable. Primary benefit of AI assistance was consistency, not speed.
		\item The five classification dimensions (layer, side, format, observability, method) make each requirement independently testable and unambiguously assigned to one implementation unit.
		\item MCP server wraps the TOML file with a typed write interface. Claude Code agents use it throughout design and verification to update status, add labels, and link files --- the narrow interface prevents silent data corruption that would occur with direct file edits.
		\item P1 = need-to-have (shall requirements, core correctness). P2 = nice-to-have. P3 = optional (should clauses). Priority drives the verification schedule.
		\item \texttt{label} and \texttt{file} are the traceability endpoints --- label names the TB procedure or RTL tag, file is the testbench or RTL source. Together they link each requirement to runnable evidence.
	\end{itemize}}

			\end{itemize}

\section{Conclusion}
\begin{frame}[t]{Conclusion and Future Work}
	\small
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\textbf{Delivered}\\[0.15em]
			Full CAN/CAN FD node in VHDL-93.
			28\,/\,37 requirements closed.
			4{,}608\,LE at 127\,MHz (Cyclone~10~LP).\\[0.5em]
			\textbf{Two lessons}
			\begin{enumerate}\setlength{\itemsep}{3pt}
				\item Requirements structure is not RTL structure ---
				      the \texttt{side} dimension shapes testbenches, not modules
				\item Layered architecture is a practical partitioning ---
				      sub-layer boundaries are testbench boundaries
			\end{enumerate}
		\end{column}
		\begin{column}{0.48\textwidth}
			\textbf{Future work}
			\begin{enumerate}\setlength{\itemsep}{3pt}
				\item \texttt{can\_llc} implementation
				\item Hardware bring-up and PCAN interoperability
				\item BRAM migration for RX frame buffer
				\item REQ-021 frame-aware error injection
				      (all four error types)
			\end{enumerate}
		\end{column}
	\end{columns}
\end{frame}
