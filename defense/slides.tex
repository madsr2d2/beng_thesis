\section{Introduction}
\begin{frame}{Controller Area Network (CAN) -- Bus Topology and ECUs}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\begin{center}
				\vspace{1.2em}
				\includegraphics[width=\textwidth,height=0.65\textheight,keepaspectratio]{images/bus_wire.jpeg}\\[0.3em]
				{\tiny CAN bus wire (source: Amazon.com)}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}
\note{
	\footnotesize
	\begin{itemize}\setlength{\itemsep}{4pt}
		\item CAN = Controller Area Network. Bosch, 1980s. Now ISO 11898-1 --- the dominant embedded bus standard.
		\item Each box = an ECU (Electronic Control Unit). CAN controller + transceiver chip. No host, no bus master.
		\item Used in: cars, trucks, marine, industrial machinery, aerospace, agriculture.
		\item Everllence: marine engine control. Sensors and actuators on one shared CAN bus on the IO-extender FPGA.
	\end{itemize}
}

% ==========================================
% SLIDE 2: Benefit 1
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\vspace{1.2em}
			\textbf{Simple and Low Cost}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Single shared bus replaces complex point-to-point wiring
				\item Reduces vehicle/machine weight, wiring errors, and costs
				\item Widely adopted: low-cost controllers, transceivers, and tooling
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Good example: NMEA 0183 (point-to-point, one talker per line) vs NMEA 2000
	(CAN-based, all devices on one bus) in marine systems --- exactly Everllence's world.
	Weight reduction in automotive: switching to CAN can save up to 20 kg of wiring harness.
}

% ==========================================
% SLIDE 3: Benefit 2
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\vspace{1.2em}
			\textbf{Easy Centralized Access}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item One connection point for diagnostics across all ECUs on the network
				\item Silent data logging --- the interface does not affect bus traffic
				\item Remote ECU firmware and configuration updates via CAN frames
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Silent mode is critical for non-invasive diagnostics in production systems ---
	a logging device that influences the bus would invalidate the measurement.
	ECU flashing over CAN is standard in automotive (UDS protocol) and is used
	in Everllence's engine platforms too.
}

% ==========================================
% SLIDE 4: Benefit 3
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\vspace{1.2em}
			\textbf{Robust Against Interference}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Differential signaling (CANH/CANL): noise couples equally to both wires and cancels
				\item Built-in error detection: bit, stuff, CRC, form, and ACK errors
				\item Faulty nodes automatically isolated --- bus off prevents one node corrupting the network
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	The marine engine bay is one of the harshest EMI environments imaginable ---
	high-current motor drives, ignition systems, inverters. Differential signaling
	and bus-off isolation are not nice-to-haves here, they are requirements.
	Bus off is implemented in the FCE (Fault Confinement Entity) in this project ---
	good segue to the architecture slides later.
}

% ==========================================
% SLIDE 5: Benefit 4
% ==========================================
\begin{frame}{Why CAN -- Key Attributes and Benefits}
	\begin{columns}[T]
		\canbusanchor
		\begin{column}{0.41\textwidth}
			\vspace{1.2em}
			\textbf{Efficient Priority-Based Access}
			\begin{itemize}\setlength{\itemsep}{10pt}
				\item Bitwise arbitration: lowest identifier wins, no collisions, no bus master
				\item Safety-critical messages always transmitted first
				\item Lower-priority frames automatically fill remaining bus bandwidth
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
\note{
	Non-destructive arbitration: the losing node does not corrupt the winning frame ---
	it simply backs off and retries. No wasted bandwidth, no collision recovery overhead.
	This is the key reason CAN outperformed competing protocols like VAN and SCP
	in the 1990s for safety-critical automotive use.
}

\section{Problem}
\begin{frame}{Why Clean-Slate}
	\begin{columns}[T]
		\begin{column}{0.53\textwidth}
			\textbf{Existing \texttt{can\_bus\_controller}: four limitations}
			\begin{itemize}
				\item Single bit rate --- no data-phase switching
				\item Dynamic stuffing only --- no fixed stuffing or SBC
				\item Single CRC engine (CRC-15) --- no CRC-17/21
				\item Coupled fault confinement --- shared error path
			\end{itemize}
			\vspace{0.5em}
			Extension to CAN FD would require replacing every major block.
		\end{column}
		\begin{column}{0.42\textwidth}
			\textbf{Everllence context}\\[0.4em]
			Marine engine control: sensors and actuators on a shared bus.
			An IO-extender FPGA needs a new CAN node.\\[0.4em]
			CAN Classic: 1\,Mbit/s, 8-byte payload --- becoming a constraint.\\[0.3em]
			CAN FD: 5\,Mbit/s data phase, 64-byte payload --- the migration path.
		\end{column}
	\end{columns}
\end{frame}
\note{
	\footnotesize
	\textbf{Third-party IP --- why rejected:} No available core satisfies all five
	constraints simultaneously.
	\begin{itemize}\setlength{\itemsep}{3pt}
		\item IP ownership: 30-year service horizon; no vendor dependency
		\item Verification authority: must own the verification, not inherit it
		\item Architectural scope: existing cores include full node (buffers, filters, DMA) --- Everllence needs protocol engine only
		\item Integration conventions: Avalon-ST interfaces, existing FPGA infrastructure
		\item Platform independence: Xilinx/Intel-locked cores are a non-starter
	\end{itemize}
}

\begin{frame}[t]{Split Path: Drift in Parallel Development}
	\texttt{side} dimension misread as implementation-facing: TX and RX split into
	separate FSMs. TX stable Feb~2026; RX stubs March~31.
	Two weeks of parallel development $\to$ \textbf{eight timing divergences.}
	Two \texttt{can\_mac\_bs} and two \texttt{can\_mac\_crc} instances;
	each SP-boundary defined twice, independently.\\[0.3em]
	\textbf{Three representative bugs --- two waveform sets each:}\\[0.2em]
	\small
	\renewcommand{\arraystretch}{1.5}
	\begin{tabular}{>{\bfseries}p{0.07\textwidth} p{0.50\textwidth} p{0.34\textwidth}}
		\toprule
		Bug & Symptom                                                                  & Trigger \\
		\midrule
		A+C & SOF structural divergence: TX 2-bit SOF, RX 1~SP behind in all CRC feeds
		    & Systematic; every FD frame                                                         \\

% ==========================================
% SLIDE 5B: Split Path — Verification Blind Spot
% ==========================================
\begin{frame}[t]{Split Path: A Verification Blind Spot}
	\textbf{Split-path approach:} tests verified the RTL against a reference model.
	The reference model was calibrated to the same implicit timing assumptions
	as the RTL --- bugs cancelled out.\\[0.8em]
	\textbf{Unified-path approach:} tests verify that a second independent CAN
	implementation agrees with the first.
	This is exactly what the protocol guarantees.\\[0.8em]
	Bug~D is structurally impossible to trigger in a single-node testbench
	regardless of frame count --- only visible with two nodes and realistic bus delay.
\end{frame}

\section{PCS}
\begin{frame}{PCS: Dual Bit Rate Switching and TDC}
	\textbf{Bit rate switching} --- arbitration: nominal (e.g.\ 500~kbit/s);
	data phase: fast (up to 5~Mbit/s), entered on BRS bit.\\[0.3em]
	\textbf{TDC} --- transceiver loop delay measured and compensated so the
	Secondary Sample Point (SSP) lands correctly for bit-error monitoring.
