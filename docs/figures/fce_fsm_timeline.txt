s_error_active("`**s_error_active**
─────────
• mac_o.error_active = 1, bus_off = 0
• TX fault → TEC +8
• TX success → TEC −1 (min 0)
• RX primary / flag error → REC +8
• RX other error → REC +1
• RX success → REC −1 (clamp to 127)`")

s_error_passive("`**s_error_passive**
─────────
• mac_o.error_active = 0, bus_off = 0
• Same TEC / REC update rules apply
• Sends passive error flag (recessive)`")

s_bus_off("`**s_bus_off**
─────────
• mac_o.bus_off = 1
• llc_o.bus_off = 1
• pcs_o.bus_off = 1
• TEC / REC updates suspended
• Count idle_condition pulses from PCS
  (each pulse = 11 consecutive recessive bits)`")

ev_recovery("`**Bus-off recovery**
─────────
• idle_count = 128
• TEC reset to 0
• REC reset to 0`")

s_error_active  -->|"TEC > 127 or REC > 127"| s_error_passive
s_error_passive -->|"TEC ≤ 127 and REC ≤ 127"| s_error_active
s_error_active  -->|"TEC > 255"| s_bus_off
s_error_passive -->|"TEC > 255"| s_bus_off
s_bus_off       --> ev_recovery
ev_recovery     -->|"return to normal operation"| s_error_active
