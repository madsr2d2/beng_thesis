%% LLC TX path (LLC is Avalon-ST source)
LLC ==>|"Avalon-ST source"| SER
SER -->|"ready"| LLC
SER ==>|"transfer_status[2:0]"| LLC

%% LLC RX path (FSM is Avalon-ST source)
FSM ==>|"Avalon-ST source"| LLC
LLC -->|"ready"| FSM

%% FSM -> PCS
FSM -->|"tx_data"| PCS
FSM -->|"transmitting"| PCS
FSM -->|"next_bit_is_brs"| PCS
FSM -->|"next_bit_is_res"| PCS
FSM -->|"data_phase_stop"| PCS
FSM -->|"do_hard_sync"| PCS

%% PCS -> FSM
PCS -->|"rx_data"| FSM
PCS -->|"sample_point"| FSM
PCS -->|"ssp"| FSM
PCS ==>|"tdc_delay[2:0]"| FSM

%% FSM -> FCE
FSM -->|"transmitting"| FCE
FSM -->|"error"| FCE
FSM -->|"primary_error"| FCE
FSM -->|"sending_error_overload_flag"| FCE
FSM -->|"passive_tx_ack_error_exempt_1"| FCE
FSM -->|"error_delimiter_too_late"| FCE
FSM -->|"successful_transfer"| FCE

%% FCE -> FSM
FCE -->|"error_active"| FSM
FCE -->|"bus_off"| FSM

%% FCE <-> PCS
FCE -->|"bus_off"| PCS
PCS -->|"idle_condition"| FCE

%% FCE <-> LLC interface
FCE -->|"bus_off"| LLC
LLC -->|"normal_mode"| FCE

%% PCS <-> Transceiver
PCS -->|"tx"| XCVR
XCVR -->|"rx"| PCS

%% User Application <-> LLC interface (TX path)
APP ==>|"Avalon-ST source (TX)"| LLC
LLC -->|"ready"| APP
LLC ==>|"transfer_status[2:0]"| APP

%% LLC interface <-> User Application (RX path)
LLC ==>|"Avalon-ST source (RX)"| APP
APP -->|"ready"| LLC

class LLC llc
class SER,FSM,BS,CRC mac
class PCS pcs
