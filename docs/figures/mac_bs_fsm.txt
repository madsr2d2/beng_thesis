s_reset("`**s_reset**
─────────
• count = 0, last_polarity = recessive
• stuff_count = 0, bs_o = reset`")


s_dynamic("`**s_dynamic**
─────────
• On valid: update last_polarity
• data ≠ last_polarity: count = 1
• data = last_polarity, count < 4: count + 1
• 5th identical bit (count = 4):
  insert ~last_polarity, count = 0
  stuff_count + 1
  SBC = Gray(stuff_count + 1) & parity`")


ev_fsb_rise("`**fsb_en rising edge**
─────────
• Emit ~last_polarity as initial FSB
• Pending SB reused if available
  (keeps TX/RX stuff_count aligned)
• count = 0`")


s_fixed("`**s_fixed**
─────────
• On valid, no pending FSB:
• count < 3: count + 1
• count = 3 (every 4 real bits):
  insert ~data, count = 0`")


ev_fsb_fall("`**fsb_en falling edge**
─────────
• Cancel pending FSB (bs_o.valid = 0)
• Resume dynamic stuffing`")

s_reset --> s_dynamic
s_dynamic -->|"rising edge fsb_en"| ev_fsb_rise
ev_fsb_rise --> s_fixed
s_fixed -->|"falling edge fsb_en"| ev_fsb_fall
ev_fsb_fall --> s_dynamic

class s_reset reset
class s_dynamic,s_fixed ff
class ev_fsb_rise,ev_fsb_fall wd4
