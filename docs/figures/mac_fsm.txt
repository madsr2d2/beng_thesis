  • Passive: drive 6 recessive`")

  s_error_delimiter -->|"dominant (error or overload)"| s_error_flag

subgraph frame_fields["`**Frame Transmission / Reception**`"]
  s_arbitration("`**s_arbitration**
  ─────────
  • SOF → ID(11) → RTR/SRR/RRS → IDE → ID(18) → RTR-ext
  • TX: drive bits
  • TX/RX: capture into llc_frame`")

  s_fdf_r1_r0("`**s_fdf_r1_r0**
  ─────────
  • TX: drive FDF/r1/r0 bit
  • RX: capture into llc_frame`")

  s_res_r0("`**s_res_r0**
  ─────────
  • TX: drive dominant`")

  s_brs("`**s_brs**
  ─────────
  • TX: drive BRS bit
  • TX/RX: set data phase if recessive`")

  s_esi("`**s_esi**
  ─────────
  • TX: recessive if error-passive or ESI flag set
  • RX: capture into llc_frame`")

  s_dlc("`**s_dlc**
  ─────────
  • TX: drive DLC bits
  • RX: capture into llc_frame and derive data_len and crc_len`")

  s_data("`**s_data**
  ─────────
  • TX: drive data bits
  • RX: capture into llc_frame`")

  s_sbc("`**s_sbc**
  ─────────
  • TX: drive SBC bits
  • RX: compare (error deferred to EOF)`")

  s_crc("`**s_crc**
  ─────────
  • TX: drive CRC bits
  • RX: compare (error deferred to EOF)`")

  s_crc_delimiter("`**s_crc_delimiter**
  ─────────
  • TX: drive recessive bit`")

  s_ack("`**s_ack**
  ─────────
  • TX: latch ack_success
  • RX: drive dominant bit if no CRC error detected `")

  s_ack_delimiter("`**s_ack_delimiter**
  ─────────
  • TX: drive recessive bit`")

  s_eof("`**s_eof**
  ─────────
  • TX: drive recessive
  • RX: trigger stream of llc_frame to LLC
  • TX/RX: signal frame complete to FCE`")

  s_arbitration --> s_fdf_r1_r0
  s_fdf_r1_r0 -->|FD or CC-ext| s_res_r0
  s_fdf_r1_r0 -->|CC-base| s_dlc
  s_res_r0 -->|FD| s_brs
  s_res_r0 -->|CC-ext| s_dlc
  s_brs --> s_esi
  s_esi --> s_dlc
  s_dlc -->|"data_len > 0, not RTR"| s_data
  s_dlc -->|FD, data_len = 0| s_sbc
  s_dlc -->|CC, data_len = 0| s_crc
  s_data -->|FD| s_sbc
  s_data -->|CC| s_crc
  s_sbc --> s_crc
  s_crc --> s_crc_delimiter
  s_crc_delimiter --> s_ack
  s_ack --> s_ack_delimiter
  s_ack_delimiter --> s_eof
end
