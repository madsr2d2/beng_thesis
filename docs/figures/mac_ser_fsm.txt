s_load_config_byte_0("`**s_load_config_byte_0**
─────────
• Idle: llc ready = 1, MAC valid = 0
• Awaiting valid + SOP from LLC
• On SOP: latch IDE, FDF, FTYP, BRS, ESI
  into llc_metadata (held for entire frame)`")

s_load_config_byte_1("`**s_load_config_byte_1**
─────────
• llc ready = 1, MAC valid = 0
• On valid: latch DLC into llc_metadata
• Init ID counters from IDE flag:
  IDE=0 → id_bits = 11, padding = 21
  IDE=1 → id_bits = 29, padding = 3`")

s_load_llc_frame_byte("`**s_load_llc_frame_byte**
─────────
• llc ready = 1
• On valid: latch byte, count = 0
• Present byte MSB as MAC tx data`")

s_shift_out_bits("`**s_shift_out_bits**
─────────
• llc ready = 0 (holds LLC back-pressure)
• Real bit (id_bits > 0 or past ID region):
  assert MAC valid, wait for MAC ready
  left-shift buffer, count + 1
• Padding bit (id_bits = 0, padding > 0):
  skip silently, no MAC valid asserted`")

ev_abort("`**Frame end / abort**
─────────
• transfer_status ≠ ongoing
• count reset to 0`")
