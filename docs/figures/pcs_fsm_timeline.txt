subgraph ctrl_path["Per-TQ control path  (runs before segment FSM)"]
  direction TB
  ev_tq("`**TQ boundary**
  ─────────
  clk_count = prescaler − 1
  rx_bus_prev ← rx_i`")

  ev_vsync("`**v_do_sync**  (edge detect)
  ─────────
  rx_bus_prev = recessive
  AND rx_i = dominant
  AND sync_applied = 0
  AND transmitting = 0`")

  hs_check("`**do_hard_sync = 1?**
  (only when v_do_sync)`")

  ev_hs("`**Hard sync action**
  ─────────
  seg_count = 0
  segment ← prop_seg
  sync_applied = 1`")

  ev_tq --> ev_vsync --> hs_check -->|"yes"| ev_hs
end

subgraph seg_fsm["Bit timing segments"]
  direction TB
  s_sync_seg("`**s_sync_seg**
  ─────────
  1 TQ fixed`")

  s_prop_seg("`**s_prop_seg**
  ─────────
  • active_prop_seg TQs
  • v_do_sync + phase_error > SJW:
    phase1_extension = SJW
  • v_do_sync + phase_error ≤ SJW:
    reset to prop_seg`")

  s_phase_seg1("`**s_phase_seg1**
  ─────────
  • active_phase_seg1
    (+phase1_extension) TQs
  • v_do_sync: same as prop_seg`")

  s_phase_seg2("`**s_phase_seg2**
  ─────────
  • active_phase_seg2
    (−phase2_shortening) TQs
  • v_do_sync + phase_error > SJW:
    phase2_shortening = SJW
  • v_do_sync + phase_error ≤ SJW:
    reset to prop_seg`")

  s_sync_seg -->|"1 TQ"| s_prop_seg
  s_prop_seg -->|"seg_count ≥ active_prop_seg − 1"| s_phase_seg1
end

subgraph io_sg["Bus drive / sample"]
  direction TB
  ev_ssp("`**SSP**
  ─────────
  secondary_sample_point = 1
  rx_data latched. transmitting only
  seg_count = phase_seg1 + ext − 2`")

  ev_sp("`**Sample Point**
  ─────────
  sample_point = 1
  rx_data latched. sync_applied = 0
  BRS recessive → data-phase timing
  data_phase_stop → nominal timing`")

  ev_bb("`**Bit Boundary**
  ─────────
  tx_o ← mac_i.tx_data
  bit_boundary = 1
  sync_applied = 0`")

  ev_ssp -.->|"1 TQ before SP"| ev_sp
end

subgraph tdc_sg["TDC pipeline  (data phase, transmitting only)"]
  direction TB
  tdc_count("`**Count recessive TQs**
  delay_count_tq + 1 per TQ
  while rx_i = recessive`")
  tdc_wait("`**Echo detected**
  rx_i = dominant
  delay_count_tq = propagation delay`")
  tdc_down("`**Countdown**
  ssp_standoff_active = 1
  delay_count_tq − 1 per TQ`")
  tdc_count -->|"rx_i = dominant"| tdc_wait
  tdc_wait -->|"ESI bit boundary"| tdc_down
end
